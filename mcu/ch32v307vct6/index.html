示例代码 · GPIO 闪烁

CH32V307VCT6 · main.c
// CH32V307VCT6 - RISC-V GPIO 闪烁
#include "ch32v30x.h"
int main(void) {
    SystemInit();
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);
    GPIO_InitTypeDef g = {0};
    g.GPIO_Pin = GPIO_Pin_0; g.GPIO_Mode = GPIO_Mode_Out_PP;
    g.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &g);
    while(1) {
        GPIO_WriteBit(GPIOA, GPIO_Pin_0, !GPIO_ReadOutputDataBit(GPIOA, GPIO_Pin_0));
        Delay_Ms(500);
    }
}
