示例代码 · GPIO 闪烁

RP2040 · main.c
// RP2040 - Pico SDK GPIO 闪烁 (双核)
#include "pico/stdlib.h"
#include "pico/multicore.h"
void core1_entry() {
    while(1) { /* 核心1任务 */ }
}
int main() {
    gpio_init(25); gpio_set_dir(25, GPIO_OUT);
    multicore_launch_core1(core1_entry);
    while(1) {
        gpio_put(25, 1); sleep_ms(500);
        gpio_put(25, 0); sleep_ms(500);
    }
}
