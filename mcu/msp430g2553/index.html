示例代码 · GPIO 闪烁

MSP430G2553 · main.c
// MSP430G2553 - GPIO 闪烁 (CCS)
#include <msp430g2553.h>
void main(void) {
    WDTCTL = WDTPW | WDTHOLD;  // 停看门狗
    P1DIR |= BIT0;              // P1.0 输出
    while(1) {
        P1OUT ^= BIT0;          // 翻转
        __delay_cycles(500000);
    }
}
