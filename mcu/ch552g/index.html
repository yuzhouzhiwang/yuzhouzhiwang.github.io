示例代码 · GPIO 闪烁

CH552G · main.c
// CH552G - 8051 GPIO 闪烁
#include "ch552.h"
void delay_ms(unsigned int ms) {
    while(ms--) { unsigned int i; for(i=0;i<1000;i++); }
}
void main() {
    P1_MOD_OC &= ~(1<<0); P1_DIR_PU |= (1<<0);  // P1.0 推挽
    while(1) {
        P1 ^= (1<<0);
        delay_ms(500);
    }
}
