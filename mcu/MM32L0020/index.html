#include "hc32f005.h"
int main(void) {

💻 标准外设库函数示例

GPIO 初始化 + 翻转
// 灵动微 MM32 标准库 - GPIO
#include "mm32l0020.h"

void GPIO_Init(void) {
    GPIO_InitTypeDef g;
    RCC_AHBPeriphClockCmd(RCC_AHBENR_GPIOA, ENABLE);
    g.GPIO_Pin  = GPIO_Pin_0;
    g.GPIO_Mode = GPIO_Mode_Out_PP;     // 推挽输出
    g.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &g);
}

int main(void) {
    GPIO_Init();
    while(1) {
        GPIO_SetBits(GPIOA, GPIO_Pin_0);
        delay_ms(500);
        GPIO_ResetBits(GPIOA, GPIO_Pin_0);
        delay_ms(500);
    }
}
UART 发送 + printf重定向
// 灵动微 MM32 标准库 - UART1
#include "mm32l0020.h"

void UART1_Init(uint32_t baud) {
    GPIO_InitTypeDef g;
    UART_InitTypeDef u;
    RCC_APB2PeriphClockCmd(RCC_APB2ENR_UART1, ENABLE);
    RCC_AHBPeriphClockCmd(RCC_AHBENR_GPIOA, ENABLE);
    // PA2=TX, PA3=RX
    g.GPIO_Pin = GPIO_Pin_2; g.GPIO_Mode = GPIO_Mode_AF_PP;
    GPIO_Init(GPIOA, &g);
    g.GPIO_Pin = GPIO_Pin_3; g.GPIO_Mode = GPIO_Mode_IN_FLOATING;
    GPIO_Init(GPIOA, &g);
    u.UART_BaudRate = baud;
    u.UART_WordLength = UART_WordLength_8b;
    u.UART_StopBits = UART_StopBits_1;
    u.UART_Parity = UART_Parity_No;
    u.UART_Mode = UART_Mode_Rx | UART_Mode_Tx;
    UART_Init(UART1, &u);
    UART_Cmd(UART1, ENABLE);
}

int fputc(int ch, FILE *f) {
    UART_SendData(UART1, (uint8_t)ch);
    while(UART_GetFlagStatus(UART1, UART_FLAG_TXE) == RESET);
    return ch;
}
