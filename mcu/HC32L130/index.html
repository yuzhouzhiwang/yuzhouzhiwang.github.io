04.5 库函数示例

💻 DDL 库函数示例

GPIO 初始化 + 翻转
// 小华 DDL 库 - GPIO 配置
#include "hc32f005.h"

void GPIO_Init(void) {
    stc_gpio_cfg_t stcGpioCfg;
    // 开启GPIO时钟
    Sysctrl_SetPeripheralGate(SysctrlPeripheralGpio, TRUE);
    // PA0 推挽输出
    stcGpioCfg.enDir = GpioDirOut;
    stcGpioCfg.enDrv = GpioDrvH;
    Gpio_Init(GpioPortA, GpioPin0, &stcGpioCfg);
}

int main(void) {
    GPIO_Init();
    while(1) {
        Gpio_SetIO(GpioPortA, GpioPin0);  // 置高
        delay1ms(500);
        Gpio_ClrIO(GpioPortA, GpioPin0);  // 置低
        delay1ms(500);
    }
}
UART 发送 + 接收中断
// 小华 DDL 库 - UART1 配置
#include "hc32f005.h"

void UART1_Init(uint32_t baud) {
    stc_uart_cfg_t stcCfg;
    Sysctrl_SetPeripheralGate(SysctrlPeripheralUart1, TRUE);
    // PA2=TX, PA3=RX
    Gpio_SetAfMode(GpioPortA, GpioPin2, GpioAf1);
    Gpio_SetAfMode(GpioPortA, GpioPin3, GpioAf1);
    stcCfg.enRunMode = UartMskMode1;
    stcCfg.stcBaud.u32Baud = baud;
    stcCfg.stcBaud.enClkDiv = UartMskBaudDivm128;
    Uart_Init(UART1, &stcCfg);
    Uart_EnableIrq(UART1, UartRxIrq);
    EnableNvic(UART1_IRQn, IrqLevel3, TRUE);
}

void UART1_SendChar(char c) {
    Uart_SendData(UART1, c);
    while(Uart_GetStatus(UART1, UartTxEmpty) == FALSE);
}
ADC 采集 + 软件延时
// 小华 DDL 库 - ADC 单次转换
#include "hc32f005.h"

uint16_t ADC_Read(uint8_t ch) {
    stc_adc_cfg_t stcAdcCfg;
    stc_adc_norm_cfg_t stcNormCfg;
    Sysctrl_SetPeripheralGate(SysctrlPeripheralAdc, TRUE);
    // PA0 模拟输入
    Gpio_SetAnalogMode(GpioPortA, GpioPin0);
    DDL_ZERO_STRUCT(stcAdcCfg);
    stcAdcCfg.enAdcMode = AdcSglMode;
    Adc_Init(&stcAdcCfg);
    DDL_ZERO_STRUCT(stcNormCfg);
    stcNormCfg.enAdcInCh = ch;
    Adc_ConfigNormMode(&stcNormCfg);
    Adc_Start();
    while(Adc_GetEocFlag() == FALSE);
    return Adc_GetNormResultData();
}
