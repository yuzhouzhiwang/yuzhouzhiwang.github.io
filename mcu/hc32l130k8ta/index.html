示例代码 · GPIO 闪烁

HC32L130K8TA · main.c
// HC32L130 - DDL库 GPIO 闪烁
#include "hc32l13x.h"
int main(void) {
    Sysctrl_SetPeripheralGate(SysctrlPeripheralGpio, TRUE);
    Gpio_InitIO(0, 0, GpioDirOut);  // P00 输出
    while(1) {
        Gpio_SetIO(0, 0);
        delay1ms(500);
        Gpio_ClrIO(0, 0);
        delay1ms(500);
    }
}
