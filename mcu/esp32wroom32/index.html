示例代码 · GPIO 闪烁

ESP32-WROOM-32 · main.c
// ESP32 - Arduino GPIO 闪烁
#define LED_PIN 2
void setup() {
  pinMode(LED_PIN, OUTPUT);
  Serial.begin(115200);
  Serial.println("ESP32 Start");
}
void loop() {
  digitalWrite(LED_PIN, HIGH);
  delay(500);
  digitalWrite(LED_PIN, LOW);
  delay(500);
}
